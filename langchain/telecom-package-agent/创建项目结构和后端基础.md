from langchain.agents import AgentExecutor
from langchain.agents.format_scratchpad import format_to_openai_function_messages
from langchain.tools.render import format_tool_to_openai_function
from langchain.prompts import ChatPromptTemplate, MessagesPlaceholder
from langchain.chat_models import ChatOpenAI

class TelecomAgent:
    def __init__(self, user_id: str, session_id: str):
        self.user_id = user_id
        self.session_id = session_id
        self.memory = ConversationMemory(session_id)
        self.tools = self._init_tools()
        self.agent = self._init_agent()
        
    def _init_tools(self):
        """初始化所有工具"""
        return [
            UsageQueryTool(),
            RecommendPackageTool(),
            GetPendingBenefitsTool(),
            ClaimBenefitTool(),
            HandleComplaintTool(),
            CheckNetworkStatusTool()
        ]
    
    def _init_agent(self):
        """初始化LangChain Agent"""
        # 1. 创建提示词模板
        prompt = ChatPromptTemplate.from_messages([
            ("system", SYSTEM_PROMPT),
            MessagesPlaceholder(variable_name="chat_history"),
            ("human", "{input}"),
            MessagesPlaceholder(variable_name="agent_scratchpad")
        ])
        
        # 2. 绑定工具
        llm = ChatOpenAI(model="gpt-3.5-turbo", temperature=0.7)
        llm_with_tools = llm.bind(
            functions=[format_tool_to_openai_function(t) for t in self.tools]
        )
        
        # 3. 创建Agent
        agent = {
            "input": lambda x: x["input"],
            "chat_history": lambda x: self.memory.get_chat_history(),
            "agent_scratchpad": lambda x: format_to_openai_function_messages(
                x["intermediate_steps"]
            )
        } | prompt | llm_with_tools
        
        # 4. 创建执行器
        return AgentExecutor(agent=agent, tools=self.tools, verbose=True)
    
    async def chat(self, message: str) -> dict:
        """处理用户消息"""
        # 1. 获取用户画像
        user_profile = await self._get_user_profile()
        
        # 2. 调用Agent
        response = await self.agent.ainvoke({
            "input": message,
            "user_profile": user_profile
        })
        
        # 3. 保存到记忆
        self.memory.add_message("human", message)
        self.memory.add_message("ai", response["output"])
        
        # 4. 判断是否需要转人工
        need_human = self._check_human_intervention(response)
        
        return {
            "response": response["output"],
            "need_human": need_human,
            "suggestions": self._generate_suggestions(response)
        }
